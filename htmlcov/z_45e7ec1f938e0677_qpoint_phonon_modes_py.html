103 def reorder_frequencies(self, 

104 *, 

105 reorder_gamma: bool = True) -> None: 

106 """ 

107 By doing a dot product of eigenvectors at adjacent q-points, 

108 determines which modes are most similar and reorders the 

109 frequencies at each q-point. This means that the same mode 

110 will have the same index across different q-points, so will 

111 be plotted as the same colour in a dispersion plot, and can 

112 be followed across q-space. 

113 

114 Parameters 

115 ---------- 

116 reorder_gamma 

117 Whether to reorder frequencies at gamma-equivalent points. 

118 If an analytical correction has been applied at the gamma 

119 points (i.e LO-TO splitting) mode assignments can be 

120 incorrect at adjacent q-points where the correction hasn't 

121 been applied. So you might not want to reorder at gamma for 

122 some materials 

123 """ 

124 n_qpts = self.n_qpts 

125 n_branches = 3*self.crystal.n_atoms 

126 qpts = self.qpts 

127 eigenvecs = self.eigenvectors 

128 

129 # Initialise map, don't reorder first q-point 

130 mode_map = np.zeros((n_qpts, n_branches), dtype=np.int32) 

131 mode_map[0] = np.arange(n_branches) 

132 

133 # Only calculate reordering if the direction hasn't 

134 # changed 

135 calc_reorder = np.concatenate(([True], np.logical_not( 

136 direction_changed(qpts)))) 

137 

138 if not reorder_gamma: 

139 gamma_i = np.where(is_gamma(qpts))[0] 

140 # Don't reorder at gamma 

141 calc_reorder[gamma_i[gamma_i > 0] - 1] = False 

142 # Or at the point after 

143 calc_reorder[gamma_i[gamma_i < n_qpts - 1]] = False 

144 

145 for i in range(1, n_qpts): 

146 # Compare eigenvectors for each mode for this q-point with 

147 # every mode for the previous q-point 

148 # Explicitly broadcast arrays with repeat and tile to ensure 

149 # correct multiplication of modes 

150 curr_evecs = eigenvecs[i, :, :, :] 

151 prev_evecs = eigenvecs[i - 1, :, :, :] 

152 current_eigenvecs = np.repeat(curr_evecs, n_branches, axis=0) 

153 prev_eigenvecs = np.tile(prev_evecs, (n_branches, 1, 1)) 

154 

155 if calc_reorder[i-1]: 

156 # Compute complex conjugated dot product of every mode 

157 # of this q-point with every mode of previous q-point, 

158 # and sum the dot products over atoms (i.e. multiply 

159 # eigenvectors elementwise, then sum over the last 2 

160 # dimensions) 

161 dots = np.absolute(np.einsum('ijk,ijk->i', 

162 np.conj(prev_eigenvecs), 

163 current_eigenvecs)) 

164 

165 # Create matrix of dot products for each mode of this 

166 # q-point with each mode of the previous q-point 

167 dot_mat = np.reshape(dots, (n_branches, n_branches)) 

168 

169 # Find greatest dot product 

170 for _ in range(n_branches): 

171 max_i = np.argmax(dot_mat) 

172 # Modes are dot_mat rows 

173 mode = int(max_i/n_branches) 

174 # Prev q-pt modes are columns 

175 prev_mode = max_i % n_branches 

176 # Ensure modes aren't mapped more than once 

177 dot_mat[mode, :] = 0 

178 dot_mat[:, prev_mode] = 0 

179 

180 prev_mode_idx = np.where(mode_map[i-1] == prev_mode)[0][0] 

181 mode_map[i, prev_mode_idx] = mode 

182 else: 

183 mode_map[i] = mode_map[i-1] 

184 

185 # Actually rearrange frequencies/eigenvectors 

186 for i in range(n_qpts): 

187 self._frequencies[i] = self._frequencies[i, mode_map[i]] 

188 evec_tmp = np.copy(self.eigenvectors[i, mode_map[i]]) 

189 self.eigenvectors[i] = evec_tmp

191 def calculate_structure_factor( 

192 self, 

193 *, 

194 scattering_lengths: IsotopeDataset = 'Sears1992', 

195 dw: DebyeWaller | None = None, 

196 ) -> StructureFactor: 

197 """ 

198 Calculate the one phonon inelastic scattering for neutrons at 

199 each q-point 

200 

201 Parameters 

202 ---------- 

203 scattering_lengths 

204 Dataset of coherent scattering length for each 

205 element in the structure. This may be provided in 4 ways: 

206 

207 - An object implementing the IsotopeData interface from 

208 euphonic.data.isotopes. 

209 

210 - An explicit dictionary of float Quantity, giving spin- and 

211 isotope-averaged coherent scattering length for each element in 

212 the structure, e.g.:: 

213 

214 {'O': 5.803*ureg('fm'), 'Zn': 5.680*ureg('fm')} 

215 

216 - A string naming an appropriate data collection packaged with 

217 Euphonic (including the default value 'Sears1992'). 

218 

219 - A string filename for a user's customised data file in the same 

220 format as those packaged with Euphonic. 

221 

222 The last two options are deprecated in favour of an improved 

223 built-in dataset and the IsotopeData interface. 

224 

225 dw 

226 Data for thermal motion effects. Typically this is computed over a 

227 converged Monkhort-Pack grid, which need not correspond to the 

228 q-points of this QpointPhononModes object. 

229 

230 Returns 

231 ------- 

232 sf 

233 An object containing the structure factor for each q-point 

234 and phonon mode 

235 

236 Notes 

237 ----- 

238 

239 This method calculates the mode-resolved (not binned in energy) 

240 coherent one-phonon neutron scattering function 

241 :math:`S(Q, \\omega_{\\mathbf{q}\\nu})` **per atom**, as defined in 

242 [1]_: 

243 

244 .. math:: 

245 

246 S(Q, \\omega_{\\mathbf{q}\\nu}) = 

247 \\frac{1}{2N_{atom}} \\ 

248 \\left\\lvert 

249 {\\sum_{\\kappa}{\\frac{b_\\kappa}{M_{\\kappa}^{1/2}{\\omega_{\\mathbf{q}\\nu}^{1/2}}} 

250 (\\mathbf{Q}\\cdot \\mathbf{e}_{\\mathbf{q}\\nu\\kappa})\\exp\\left(i\\mathbf{Q}{\\cdot}\\mathbf{r}_{\\kappa}\\right) 

251 \\exp\\left(-W_{\\kappa}\\right)}} 

252 \\right\\rvert^2 

253 

254 Where :math:`\\nu` runs over phonon modes, :math:`\\kappa` runs 

255 over atoms, :math:`b_\\kappa` is the coherent neutron 

256 scattering length, :math:`M_{\\kappa}` is the atom mass, 

257 :math:`r_{\\kappa}` is the vector from the origin to atom 

258 :math:`\\kappa` in the unit cell, 

259 :math:`\\mathbf{e}_{\\mathbf{q}\\nu\\kappa}` are the normalised 

260 Cartesian eigenvectors, :math:`\\omega_{\\mathbf{q}\\nu}` are 

261 the frequencies and :math:`\\exp\\left(-W_\\kappa\\right)` is the 

262 Debye-Waller factor. :math:`N_{atom}` is the number of atoms in 

263 the unit cell, so the returned structure factor is **per atom** 

264 of sample. 

265 

266 .. [1] M.T. Dove, Structure and Dynamics, Oxford University Press, Oxford, 2003, 225-226 

267 

268 """ # noqa: E501 

269 

270 isotope_data = _get_isotope_data(scattering_lengths) 

271 sl = isotope_data.get_array( 

272 self.crystal, 'coherent_scattering_length', 

273 ).to('bohr').magnitude 

274 

275 # Calculate normalisation factor 

276 norm_factor = sl / np.sqrt(self.crystal._atom_mass) 

277 

278 # Calculate the exp factor for all atoms and qpts. atom_r is in 

279 # fractional coords, so Qdotr = 2pi*qh*rx + 2pi*qk*ry... 

280 exp_factor = np.exp(1J*2*math.pi*np.einsum( 

281 'ij,kj->ik', self.qpts, self.crystal.atom_r)) 

282 

283 # Eigenvectors are in Cartesian so need to convert hkl to 

284 # Cartesian by computing dot with hkl and reciprocal lattice 

285 recip = self.crystal.reciprocal_cell.to('1/bohr').magnitude 

286 Q = self.qpts @ recip 

287 

288 # Calculate dot product of Q and eigenvectors for all branches 

289 # atoms and q-points 

290 eigenv_dot_q = np.einsum('ijkl,il->ijk', np.conj(self.eigenvectors), Q) 

291 

292 # Calculate Debye-Waller factors 

293 temperature = None 

294 if dw: 

295 temperature = dw.temperature 

296 if dw.crystal.n_atoms != self.crystal.n_atoms: 

297 msg = format_error( 

298 'Incompatible types (mismatched atom counts).', 

299 reason=( 

300 'The DebyeWaller object used as dw is not ' 

301 'compatible with the QPointPhononModes object.'), 

302 fix='Ensure both objects have the same number of atoms.', 

303 ) 

304 raise ValueError(msg) 

305 dw_factor = np.exp(-np.einsum('jkl,ik,il->ij', 

306 dw._debye_waller, Q, Q)) 

307 exp_factor *= dw_factor 

308 

309 # Multiply Q.eigenvector, exp factor and normalisation factor 

310 term = np.einsum('ijk,ik,k->ij', eigenv_dot_q, exp_factor, norm_factor) 

311 

312 # Take mod squared and divide by frequency to get intensity 

313 sf = np.real( 

314 np.absolute(term*np.conj(term))/np.absolute(self._frequencies)) 

315 sf /= 2*self.crystal.n_atoms 

316 

317 return StructureFactor( 

318 self.crystal, self.qpts, self.frequencies, 

319 sf*ureg('bohr**2').to('mbarn'), 

320 temperature=temperature)

322 def calculate_debye_waller( 

323 self, 

324 temperature: Quantity, 

325 *, 

326 frequency_min: Quantity = Quantity(0.01, 'meV'), 

327 symmetrise: bool = True) -> DebyeWaller: 

328 """ 

329 Calculate the 3 x 3 Debye-Waller exponent for each atom over the 

330 q-points contained in this object 

331 

332 Parameters 

333 ---------- 

334 temperature 

335 Scalar float Quantity. The temperature to use in the 

336 Debye-Waller calculation 

337 frequency_min 

338 Scalar float Quantity in energy units. Excludes frequencies below 

339 this limit from the calculation, as the calculation contains a 

340 1/frequency factor which would result in infinite values. This 

341 also allows negative frequencies to be excluded 

342 symmetrise 

343 Whether to symmetrise the Debye-Waller factor based on the 

344 crystal symmetry operations. Note that if the Debye-Waller 

345 exponent is not symmetrised, the results may not be the 

346 same for unfolded and symmetry-reduced q-point grids 

347 

348 Returns 

349 ------- 

350 dw 

351 An object containing the 3x3 Debye-Waller exponent matrix 

352 for each atom 

353 

354 Notes 

355 ----- 

356 

357 As part of the structure factor calculation, 

358 :math:`\\exp\\left(W_\\kappa\\right)` is the anisotropic 

359 Debye-Waller factor for atom :math:`\\kappa`, and the exponent can 

360 be written as: 

361 

362 .. math:: 

363 

364 W_\\kappa = 

365 \\sum\\limits_{\\alpha\\beta}W^{\\alpha\\beta}_{\\kappa} 

366 Q_{\\alpha}Q_{\\beta} 

367 

368 This function calculates the :math:`W^{\\alpha\\beta}_{\\kappa}` 

369 part of the exponent, which is a 

370 :math:`N_\\mathrm{atom}\\times3\\times3` matrix that is 

371 independent of Q, so for efficiency can be precalculated to be 

372 used in the structure factor calculation. The Debye-Waller 

373 exponent matrix is calculated by [2]_: 

374 

375 .. math:: 

376 

377 W^{\\alpha\\beta}_{\\kappa} = 

378 \\frac{\\hbar}{4M_{\\kappa}\\sum\\limits_{\\mathbf{q}}{\\mathrm{weight}_\\mathbf{q}}} 

379 \\sum\\limits_{\\mathbf{q}\\nu \\in{BZ}}\\mathrm{weight}_\\mathbf{q}\\frac{e_{\\mathbf{q}\\nu\\kappa\\alpha}e^{*}_{\\mathbf{q}\\nu\\kappa\\beta}} 

380 {\\omega_{\\mathbf{q}\\nu}} 

381 \\mathrm{coth}\\left(\\frac{\\hbar\\omega_{\\mathbf{q}\\nu}}{2k_BT}\\right) 

382 

383 Where the sum is over q-points and modes :math:`\\nu` in the 

384 first Brillouin Zone (BZ), :math:`\\kappa` runs over atoms, 

385 :math:`\\alpha,\\beta` run over the Cartesian directions, 

386 :math:`M_{\\kappa}` is the atom mass, 

387 :math:`e_{q\\nu\\kappa\\alpha}` are the scalar components of the 

388 normalised Cartesian eigenvectors, :math:`\\omega_{\\mathbf{q}\\nu}` 

389 are the frequencies, and :math:`\\mathrm{weight}_\\mathbf{q}` 

390 is the per q-point symmetry weight (if the q-points are not 

391 symmetry-reduced, all weights will be equal). 

392 

393 .. [2] G.L. Squires, Introduction to the Theory of Thermal Neutron Scattering, Dover Publications, New York, 1996, 34-37 

394 """ # noqa: E501 

395 

396 # Convert units 

397 k_B = (1 * ureg.k).to('hartree/K').magnitude 

398 n_atoms = self.crystal.n_atoms 

399 atom_mass = self.crystal._atom_mass 

400 freqs = self._frequencies 

401 freq_min = frequency_min.to('hartree').magnitude 

402 qpts = self.qpts 

403 evecs = self.eigenvectors 

404 weights = self.weights 

405 temp = temperature.to('K').magnitude 

406 

407 mass_term = 1/(4*atom_mass) 

408 

409 # Mask out frequencies below frequency_min 

410 freq_mask = np.ones(freqs.shape) 

411 freq_mask[freqs < freq_min] = 0 

412 

413 if temp > 0: 

414 x = freqs/(2 * k_B * temp) 

415 freq_term = 1/(freqs*np.tanh(x)) 

416 else: 

417 freq_term = 1/(freqs) 

418 dw = np.zeros((n_atoms, 3, 3)) 

419 # Calculating the e.e* term is expensive, do in chunks 

420 chunk = 1000 

421 for i in range(int((len(qpts) - 1)/chunk) + 1): 

422 qi = i*chunk 

423 qf = min((i + 1)*chunk, len(qpts)) 

424 

425 evec_term = np.real( 

426 np.einsum('ijkl,ijkm->ijklm', 

427 evecs[qi:qf], 

428 np.conj(evecs[qi:qf]))) 

429 

430 dw += (np.einsum('i,k,ij,ij,ijklm->klm', 

431 weights[qi:qf], mass_term, freq_term[qi:qf], 

432 freq_mask[qi:qf], evec_term)) 

433 

434 dw = dw/np.sum(weights) 

435 if symmetrise: 

436 dw_tmp = np.zeros(dw.shape) 

437 (rot, trans, 

438 eq_atoms) = self.crystal.get_symmetry_equivalent_atoms() 

439 cell_vec = self.crystal._cell_vectors 

440 recip_vec = self.crystal.reciprocal_cell.to('1/bohr').magnitude 

441 rot_cart = np.einsum('ijk,jl,km->ilm', 

442 rot, cell_vec, recip_vec)/(2*np.pi) 

443 for s in range(len(rot)): 

444 dw_tmp[eq_atoms[s]] += np.einsum('ij,kjl,ml->kim', 

445 rot_cart[s], dw, rot_cart[s]) 

446 dw = dw_tmp/len(rot) 

447 

448 dw = dw*ureg('bohr**2').to(self.crystal.cell_vectors_unit + '**2') 

449 return DebyeWaller(self.crystal, dw, temperature)

451 def calculate_pdos( 

452 self, 

453 dos_bins: Quantity, 

454 *, 

455 mode_widths: Quantity | None = None, 

456 mode_widths_min: Quantity = Quantity(0.01, 'meV'), 

457 adaptive_method: AdaptiveMethod = 'reference', 

458 adaptive_error: float = 0.01, 

459 weighting: str | None = None, 

460 cross_sections: IsotopeDataset = 'BlueBook', 

461 ) -> Spectrum1DCollection: 

462 """ 

463 Calculates partial density of states for each atom in the unit 

464 cell 

465 

466 Parameters 

467 ---------- 

468 dos_bins 

469 Shape (n_e_bins + 1,) float Quantity. The energy bin edges 

470 to use for calculating the DOS. 

471 mode_widths 

472 Shape (n_qpts, n_branches) float Quantity in energy units. 

473 The broadening width for each mode at each q-point, for 

474 adaptive broadening. 

475 mode_widths_min 

476 Scalar float Quantity in energy units. Sets a lower limit on 

477 the mode widths, as mode widths of zero will result in 

478 infinitely sharp peaks. 

479 adaptive_method 

480 String. Specifies whether to use slow, reference adaptive method or 

481 faster, approximate method. Allowed options are 'reference' 

482 or 'fast', default is 'reference'. 

483 adaptive_error 

484 Scalar float. Acceptable error for gaussian approximations 

485 when using the fast adaptive method, defined as the absolute 

486 difference between the areas of the true and approximate gaussians 

487 weighting 

488 One of {'coherent', 'incoherent', 'coherent-plus-incoherent'}. 

489 If provided, produces a neutron-weighted DOS, weighted by 

490 either the coherent, incoherent, or sum of coherent and 

491 incoherent neutron scattering cross-sections. 

492 cross_sections 

493 A dataset of cross-sections for each element in the structure: 

494 this may be an object implementing the 

495 euphonic.data.isotopes.IsotopeData protocol, a string specifying a 

496 dataset, or a dictionary explicitly giving the cross-sections for 

497 each element. 

498 

499 If cross_sections is a string, it identifies a built-in or external 

500 JSON file with reference data. This collection must contain the 

501 'coherent_cross_section' or 'incoherent_cross_section' physical 

502 property, depending on the ``weighting`` argument. If ``weighting`` 

503 is None, this string argument is not used. 

504 

505 If cross sections is a dictionary, the ``weighting`` argument is 

506 ignored, and these cross-sections are used directly to calculate 

507 the neutron-weighted DOS. It must contain a key for each element 

508 in the structure, and each value must be a Quantity in the 

509 appropriate units, e.g:: 

510 

511 {'La': 8.0*ureg('barn'), 'Zr': 9.5*ureg('barn')} 

512 

513 Returns 

514 ------- 

515 

516 dos 

517 A collection of spectra, with the energy bins on the x-axis 

518 and PDOS for each atom in the unit cell on the y-axis, 

519 metadata describes the index and species of each PDOS. If 

520 the PDOS is not neutron-weighted, it is in units of modes 

521 per energy unit per atom, such that the integrated area of 

522 the total DOS is equal to 3. If the PDOS is weighted by the 

523 neutron cross-section, it is in units of area per unit 

524 energy per atom, see Notes for details. 

525 

526 Notes 

527 ----- 

528 

529 The PDOS is calculated as: 

530 

531 .. math:: 

532 

533 PDOS_{\\kappa}(\\omega) = \\frac{1}{N_\\mathrm{atom}\\sum\\limits_{\\mathbf{q}}{\\mathrm{weight}_{\\mathbf{q}}}} 

534 \\sum_{\\mathbf{q}\\nu\\alpha\\in{BZ}} 

535 \\mathrm{weight}_{\\mathbf{q}} {e_{\\mathbf{q}\\nu\\kappa\\alpha}e^{*}_{\\mathbf{q}\\nu\\kappa\\alpha}} 

536 \\delta(\\omega - \\omega_{\\mathbf{q}\\nu}) 

537 

538 Where :math:`N_\\mathrm{atom}` is the number of atoms in the unit cell, 

539 :math:`\\mathrm{weight}_\\mathbf{q}` is the per q-point weight, 

540 :math:`e_{\\mathbf{q}\\nu\\kappa\\alpha}` is the normalised 

541 eigenvector at :math:`\\mathbf{q}` for mode :math:`\\nu`, atom 

542 :math:`\\kappa` in Cartesian direction :math:`\\alpha` and 

543 :math:`\\omega_{\\mathbf{q}\\nu}` is the phonon frequency at 

544 :math:`\\mathbf{q}` for mode :math:`\\nu`, and the sum 

545 :math:`BZ` is over the 1st Brillouin Zone. 

546 

547 The neutron-weighted PDOS is calculated as: 

548 

549 .. math:: 

550 

551 PDOS^\\mathrm{neutron}_{\\kappa}(\\omega) = \\frac{\\sigma_{\\kappa}M_\\mathrm{avg}}{M_\\kappa} PDOS_{\\kappa}(\\omega) 

552 

553 Where :math:`\\sigma_\\kappa` is the neutron scattering cross 

554 section, :math:`M_\\mathrm{avg}` is the average mass of atoms in the 

555 unit cell and :math:`M_\\kappa` is the mass of atom 

556 :math:`\\kappa`, so the neutron-weighted PDOS is returned in 

557 units of area per unit energy. 

558 

559 """ # noqa: E501 

560 

561 cs = _get_cross_sections(self.crystal, weighting, cross_sections) 

562 

563 if cs is not None: 

564 ex_units = '[length]**2' 

565 if not cs[0].check(ex_units): 

566 msg = format_error( 

567 'Unexpected dimensions in cross_sections units ' 

568 f'({cs[0].dimensionality}).', 

569 fix=f'cross_sections should have units of {ex_units}.', 

570 ) 

571 raise ValueError(msg) 

572 

573 cs = cs.to('mbarn') 

574 

575 evec_weights = np.real(np.einsum('ijkl,ijkl->ijk', 

576 self.eigenvectors, 

577 np.conj(self.eigenvectors))) 

578 # Normalise DOS to per atom 

579 evec_weights /= self.crystal.n_atoms 

580 crystal = self.crystal 

581 for i in range(crystal.n_atoms): 

582 dos = self._calculate_dos(dos_bins, mode_widths=mode_widths, 

583 mode_widths_min=mode_widths_min, 

584 adaptive_method=adaptive_method, 

585 adaptive_error=adaptive_error, 

586 mode_weights=evec_weights[:, :, i]) 

587 if cs is not None: 

588 # Neutron weighted DOS is per atom of sample 

589 avg_atom_mass = np.mean(crystal._atom_mass) 

590 dos *= cs[i]*avg_atom_mass/crystal._atom_mass[i] 

591 if i == 0: 

592 all_dos_y_data = np.zeros(( 

593 crystal.n_atoms, len(dos_bins) - 1))*dos.units 

594 all_dos_y_data[i] = dos 

595 metadata = {'line_data': 

596 [{'species': symbol, 'index': idx} 

597 for idx, symbol in enumerate(crystal.atom_type)]} 

598 if weighting is not None: 

599 metadata['weighting'] = weighting 

600 

601 return Spectrum1DCollection( 

602 dos_bins, all_dos_y_data, metadata=metadata)
