100 symbol_counts = Counter(symbols)

117def _remove_breaks(distances: np.ndarray, btol: float = 10.) -> list[int]: 

118 """Collapse large breaks in cumulative-distance array 

119 

120 These correspond to discontinuous regions of the x-axis: in euphonic 

121 plotting this is usually handled by splitting the spectrum and plotting 

122 to new axes, but Phonon Website does not handle this. 

123 

124 Data is modified in-place 

125 

126 A list of identified breakpoints is returned 

127 

128 """ 

129 diff = np.diff(distances) 

130 median = np.median(diff) 

131 breakpoints = np.where((diff / median) > btol)[0] + 1 

132 

133 for line_break in reversed(breakpoints): 

134 distances[line_break:] -= diff[line_break - 1] 

135 

136 return breakpoints.tolist()

139def _find_duplicates(distances: np.ndarray) -> list[int]: 

140 """Identify breakpoints where a q-point is repeated in list 

141 

142 Return the higher index of two points: this works more nicely with phonon 

143 website interpretation of line break points. 

144 """ 

145 

146 diff = np.diff(distances) 

147 duplicates = np.where(np.isclose(diff, 0.))[0] + 1 

148 

149 return duplicates.tolist()

152def _combine_neighbouring_labels(x_tick_labels: XTickLabels) -> XTickLabels: 

153 """Merge neighbouring labels in x_tick_label list 

154 

155 If labels are the same, only keep one. 

156 

157 If labels are different, join with | 

158 

159 e.g.:: 

160 

161 [(1, "X"), (2, "X"), (4, "A"), (7, "Y"), (8, "Z")] 

162 

163 --> 

164 

165 [(2, "X"), (4, "A"), (8, "Y|Z")] 

166 

167 """ 

168 labels = dict(x_tick_labels) 

169 

170 for (index, next_index) in pairwise(sorted(labels)): 

171 

172 if index + 1 == next_index: # Labels are neighbours 

173 

174 if labels[index] != labels[next_index]: # Combine differing labels 

175 labels[next_index] = f'{labels[index]}|{labels[next_index]}' 

176 

177 del labels[index] # Drop redundant label 

178 

179 return sorted(labels.items())

182def _modes_to_phonon_website_dict( 

183 modes: QpointPhononModes, 

184 name: str | None = None, 

185 repetitions: tuple[int, int, int] = (2, 2, 2), 

186 x_tick_labels: XTickLabels | None = None, 

187) -> PhononWebsiteData: 

188 """Convert QpointPhononModes to dict of phonon-website-friendly data""" 

189 

190 qpts = modes.qpts 

191 eigenvectors = modes.eigenvectors 

192 

193 abscissa = _calc_abscissa(modes.crystal.reciprocal_cell, qpts) 

194 

195 duplicates = _find_duplicates(abscissa) 

196 breakpoints = _remove_breaks(abscissa) 

197 

198 breakpoints = sorted({0, *duplicates, *breakpoints, len(abscissa)}) 

199 line_breaks = list(pairwise(breakpoints)) 

200 

201 if x_tick_labels is None: 201 ↛ 205line 201 didn't jump to line 205 because the condition on line 201 was always true

202 x_tick_labels = get_qpoint_labels(qpts, 

203 cell=modes.crystal.to_spglib_cell()) 

204 

205 x_tick_labels = [(int(key) + 1, str(value)) 

206 for key, value in x_tick_labels] 

207 x_tick_labels = _combine_neighbouring_labels(x_tick_labels) 

208 

209 mass_weights = 1 / np.sqrt(modes.crystal.atom_mass) 

210 vectors = eigenvectors * mass_weights[None, None, :, None] 

211 # Convert complex numbers to a final axis over (real, imag) 

212 vectors = vectors.view(float).reshape(*eigenvectors.shape[:-1], 3, 2) 

213 

214 return PhononWebsiteData( 

215 name=name, 

216 **_crystal_website_data(modes.crystal), 

217 highsym_qpts=x_tick_labels, 

218 distances=abscissa.magnitude.tolist(), 

219 qpoints=modes.qpts.tolist(), 

220 eigenvalues=modes.frequencies.to('1/cm').magnitude.tolist(), 

221 vectors=vectors.tolist(), 

222 repetitions=repetitions, 

223 line_breaks=line_breaks, 

224 )
